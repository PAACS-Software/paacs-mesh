cmake_minimum_required(VERSION 3.30...4.4)
project(PAACSConsumer LANGUAGES C CXX)
find_package(PAACSMesh 1.2.0 EXACT CONFIG REQUIRED OPTIONAL_COMPONENTS OCCT)
add_executable(cpp-consumer main.cpp)
target_link_libraries(cpp-consumer PRIVATE PAACS::mesh)
add_executable(c-consumer main.c)
set_target_properties(c-consumer PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(c-consumer PRIVATE PAACS::mesh)
enable_testing()
add_executable(background-jobs-consumer ../background_jobs.cpp)
target_link_libraries(background-jobs-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.background_jobs COMMAND background-jobs-consumer)
if(TARGET PAACS::ui)
    add_executable(ui-consumer ui.c)
    set_target_properties(ui-consumer PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(ui-consumer PRIVATE PAACS::ui)
    add_test(NAME consumer.ui COMMAND ui-consumer)
endif()
add_test(NAME consumer.cpp COMMAND cpp-consumer)
add_test(NAME consumer.c COMMAND c-consumer)
add_executable(geometry-consumer ../geometry.cpp)
target_link_libraries(geometry-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.geometry COMMAND geometry-consumer)
if(TARGET PAACS::occt)
    add_executable(cad-consumer cad.cpp)
    target_link_libraries(cad-consumer PRIVATE PAACS::occt)
    add_test(NAME consumer.cad COMMAND cad-consumer
        "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/cad/through_hole.step")
    add_executable(native-cad-consumer native_cad.cpp)
    target_link_libraries(native-cad-consumer PRIVATE PAACS::occt)
    add_test(NAME consumer.native_cad COMMAND native-cad-consumer
        "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/cad/cylinder.step")
endif()
add_executable(graded-consumer ../graded_geometry.cpp)
target_link_libraries(graded-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.graded COMMAND graded-consumer)
add_executable(primitives-consumer ../primitives.cpp)
target_link_libraries(primitives-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.primitives COMMAND primitives-consumer)

add_executable(primitive-elements-consumer ../primitive_elements.cpp)
target_link_libraries(primitive-elements-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.primitive_elements COMMAND primitive-elements-consumer)
add_executable(uniform-features-consumer ../uniform_features.cpp)
target_link_libraries(uniform-features-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.uniform_features COMMAND uniform-features-consumer)
add_executable(local-controls-consumer ../local_controls.cpp)
target_link_libraries(local-controls-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.local_controls COMMAND local-controls-consumer)
add_executable(profiled-controls-consumer ../profiled_controls.cpp)
target_link_libraries(profiled-controls-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.profiled_controls COMMAND profiled-controls-consumer)
add_executable(complex-controls-consumer ../complex_geometry_controls.cpp)
target_link_libraries(complex-controls-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.complex_controls COMMAND complex-controls-consumer)
add_executable(spatial-controls-consumer ../spatial_controls.cpp)
target_link_libraries(spatial-controls-consumer PRIVATE PAACS::mesh)
add_test(NAME consumer.spatial_controls COMMAND spatial-controls-consumer)
